// Core Execution Processor - Async
async function evaluateState({ telemetry }) {
  if (!telemetry) return null;

  const [state] = await dbQuery`
    INSERT INTO active_states (metrics)
    VALUES (${telemetry})
    RETURNING id, metrics, timestamp
  `;

  const trajectory = await dbQuery`
    SELECT id, vectors, timestamp
    FROM historical_states
    ORDER BY timestamp DESC
  `;

  return { state, trajectory };
}
